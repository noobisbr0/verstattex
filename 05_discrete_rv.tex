\section{Дискретные случайные величины}

\subsection{Определение и закон распределения}

\begin{definition}
\textbf{Случайной величиной}~$X$ называется числовая функция, ставящая в соответствие каждому элементарному исходу $\omega \in \Omega$ некоторое действительное число $X(\omega)$.
\end{definition}

Случайная величина называется \textbf{дискретной}, если множество ее возможных значений конечно или счетно. Закон ее распределения задается таблицей (рядом распределения), где перечислены значения $x_i$ и соответствующие им вероятности $p_i = \Prob(X = x_i)$. Сумма всех вероятностей обязана равняться единице: $\sum p_i = 1$.

\subsection{Математическое ожидание и его свойства}

\begin{definition}
\textbf{Математическое ожидание} $\E[X]$ дискретной случайной величины~--- это сумма произведений всех ее возможных значений на их вероятности:
\[
\E[X] = \sum_{i} x_i p_i.
\]
\end{definition}

\begin{theorem}[Свойства математического ожидания]\label{thm:ev_properties}
Пусть $X$ и $Y$~--- случайные величины, а $c, a, b$~--- константы.
\begin{enumerate}
\item $\E[c] = c$.
\item $\E[aX + b] = a \E[X] + b$.
\item $\E[XY] = \E[X] \E[Y]$ (только если $X$ и $Y$ независимы).
\end{enumerate}
\end{theorem}

\begin{proof}
1. Константу~$c$ можно рассматривать как дискретную случайную величину, принимающую значение~$c$ с вероятностью $p_1 = 1$. Тогда $\E[c] = c \cdot 1 = c$.

2. Рассмотрим величину $Z = aX + b$. Ее значения равны $(a x_i + b)$ с вероятностями $p_i$:
\[
\E[aX + b] = \sum_{i} (a x_i + b) p_i = a \left( \sum_{i} x_i p_i \right) + b \left( \sum_{i} p_i \right) = a \E[X] + b \cdot 1.
\]

3. Доказательство для произведения приведено в Главе~6.
\end{proof}

\subsection{Дисперсия и ее свойства}

\begin{definition}
\textbf{Дисперсия} $\Var[X]$~--- это математическое ожидание квадрата отклонения случайной величины от ее математического ожидания:
\[
\Var[X] = \E\left[(X - \E[X])^2\right].
\]
\end{definition}

\begin{theorem}[Вычислительная формула дисперсии]
Дисперсия равна разности математического ожидания квадрата величины и квадрата ее математического ожидания: $\Var[X] = \E[X^2] - (\E[X])^2$.
\end{theorem}

\begin{proof}
Обозначим $\mu = \E[X]$. Тогда $\Var[X] = \E\left[(X - \mu)^2\right]$. Раскроем квадрат:
\[
\Var[X] = \E[X^2 - 2\mu X + \mu^2] = \E[X^2] - \E[2\mu X] + \E[\mu^2].
\]
Используя свойства математического ожидания, получаем:
\[
\Var[X] = \E[X^2] - 2\mu \E[X] + \mu^2 = \E[X^2] - 2\mu^2 + \mu^2 = \E[X^2] - (\E[X])^2.
\]
\end{proof}

\begin{theorem}[Свойства дисперсии]\label{thm:var_properties}
\begin{enumerate}
\item $\Var[c] = 0$.
\item $\Var[X + c] = \Var[X]$.
\item $\Var[aX] = a^2 \Var[X]$.
\end{enumerate}
\end{theorem}

\begin{proof}
\begin{enumerate}
\item $\Var[c] = \E[c^2] - (\E[c])^2 = c^2 - c^2 = 0$. 
\item $\Var[X + c] = \E\left[((X+c) - \E[X+c])^2\right] = \E\left[(X + c - \E[X] - c)^2\right] = \E\left[(X - \E[X])^2\right] = \Var[X]$. 
\item $\Var[aX] = \E\left[(aX - \E[aX])^2\right] = \E\left[(aX - a\E[X])^2\right] = \E\left[a^2 (X - \E[X])^2\right] = a^2 \Var[X]$.
\end{enumerate}
\end{proof}

\begin{example}[Расчет характеристик дискретной величины]
Пусть дискретная случайная величина~$X$ задана рядом распределения:
\begin{center}
\begin{tabular}{c|ccc}
\toprule
$X$ & $-1$ & $0$ & $2$ \\
\midrule
$P$ & $0{,}2$ & $0{,}3$ & $0{,}5$ \\
\bottomrule
\end{tabular}
\end{center}
Найдем математическое ожидание $\E[X]$ и дисперсию $\Var[X]$.

\textbf{Решение:}
1. Вычислим математическое ожидание:
\[
\E[X] = -1 \cdot 0{,}2 + 0 \cdot 0{,}3 + 2 \cdot 0{,}5 = -0{,}2 + 0 + 1 = 0{,}8.
\]
2. Для вычисления дисперсии по формуле $\Var[X] = \E[X^2] - (\E[X])^2$ найдем $\E[X^2]$. Значения $X^2$ равны $1, 0, 4$ с теми же вероятностями:
\[
\E[X^2] = (-1)^2 \cdot 0{,}2 + 0^2 \cdot 0{,}3 + 2^2 \cdot 0{,}5 = 1 \cdot 0{,}2 + 0 + 4 \cdot 0{,}5 = 0{,}2 + 2 = 2{,}2.
\]
3. Найдем дисперсию:
\[
\Var[X] = 2{,}2 - (0{,}8)^2 = 2{,}2 - 0{,}64 = 1{,}56.
\]
\end{example}

\subsection{Графическое представление закона распределения}

Для наглядного представления дискретной случайной величины используются две основные графические формы: **многоугольник (полигон) распределения** и **функция распределения**.

\subsubsection{Многоугольник распределения вероятностей}
В декартовой системе координат по оси абсцисс откладывают возможные значения $x_i$, а по оси ординат~--- соответствующие им вероятности $p_i$. Точки $(x_i, p_i)$ соединяются отрезками прямых.

Физический смысл математического ожидания $\E[X]$ заключается в том, что оно представляет собой координату **центра тяжести** системы материальных точек с массами $p_i$, расположенных в точках $x_i$.

\subsubsection{Функция распределения (CDF)}

\begin{definition}
\textbf{Функцией распределения} случайной величины~$X$ называется функция $F(x)$, равная вероятности того, что величина~$X$ примет значение, меньшее или равное~$x$:
\[
F(x) = \Prob(X \le x).
\]
\end{definition}

Для дискретной величины функция $F(x)$ является **кусочно-постоянной (ступенчатой)** со скачками в точках $x_i$, причем величина каждого скачка в точности равна вероятности соответствующего значения $p_i = \Prob(X = x_i)$.

Для случайной величины из предыдущего примера:
\[
F(x) = \begin{cases} 
0, & x < -1, \\ 
0{,}2, & -1 \le x < 0, \\ 
0{,}5, & 0 \le x < 2, \\ 
1{,}0, & x \ge 2. 
\end{cases}
\]

\subsection{Геометрическое распределение}
Если проводятся независимые испытания с вероятностью успеха~$p$ ($q = 1 - p$), то число испытаний~$X$ до первого успеха имеет геометрическое распределение. Вероятность того, что первый успех наступит ровно в $k$-ом испытании, равна:
\[
\Prob(X = k) = q^{k-1} p, \quad k = 1, 2, 3, \dots
\]

Вероятность того, что успех наступит не позднее $n$-го испытания, можно посчитать через сумму геометрической прогрессии:
\[
\Prob(X \le n) = \sum_{k=1}^n q^{k-1} p = 1 - q^n.
\]
