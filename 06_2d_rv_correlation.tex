\section{Двумерные случайные величины и корреляция}

\subsection{Совместное распределение и математическое ожидание суммы}

Многие эксперименты требуют изучения сразу двух параметров $X$ и $Y$. Совместное распределение пары дискретных случайных величин задается матрицей вероятностей $p_{ij} = \Prob(X = x_i, Y = y_j)$.

\begin{theorem}[Линейность математического ожидания для суммы]
Для любых двух случайных величин $X$ и $Y$ (независимо от того, зависимы они или нет) выполняется:
\[
\E[X + Y] = \E[X] + \E[Y].
\]
\end{theorem}

\begin{proof}
По определению математического ожидания функции от двумерной случайной величины:
\begin{align*}
\E[X + Y] &= \sum_{i} \sum_{j} (x_i + y_j) p_{ij} = \sum_{i} \sum_{j} (x_i p_{ij} + y_j p_{ij}) \\
&= \sum_{i} x_i \left( \sum_{j} p_{ij} \right) + \sum_{j} y_j \left( \sum_{i} p_{ij} \right).
\end{align*}
Заметим, что суммирование совместной вероятности $p_{ij}$ по всем $j$ дает маргинальную (одномерную) вероятность $\Prob(X = x_i)$. Аналогично, сумма по $i$ дает $\Prob(Y = y_j)$:
\[
\E[X + Y] = \sum_{i} x_i \Prob(X = x_i) + \sum_{j} y_j \Prob(Y = y_j) = \E[X] + \E[Y].
\]
\end{proof}

\begin{theorem}[Математическое ожидание произведения независимых СВ]
Если $X$ и $Y$ независимы, то $\E[XY] = \E[X]\E[Y]$.
\end{theorem}

\begin{proof}
Если $X$ и $Y$ независимы, то вероятность совместного наступления равна произведению вероятностей: $p_{ij} = \Prob(X = x_i) \cdot \Prob(Y = y_j)$.
\[
\E[XY] = \left( \sum_i x_i \Prob(X = x_i) \right) \cdot \left( \sum_j y_j \Prob(Y = y_j) \right) = \E[X] \cdot \E[Y].
\]
\end{proof}

\subsection{Ковариация и корреляция}

\newpage
\begin{definition}
\textbf{Ковариацией} случайных величин $X$ и $Y$ называется математическое ожидание произведения их отклонений от своих математических ожиданий:
\[
\Cov(X, Y) = \E\left[(X - \E[X])(Y - \E[Y])\right].
\]
\end{definition}

\begin{theorem}[Вычислительная формула и независимость]
\leavevmode
\begin{enumerate}
\item $\Cov(X, Y) = \E[XY] - \E[X]\E[Y]$.
\item Если $X$ и $Y$ независимы, то $\Cov(X, Y) = 0$.
\end{enumerate}
\end{theorem}

\begin{theorem}[Дисперсия суммы]
\[
\Var[X + Y] = \Var[X] + \Var[Y] + 2\,\Cov(X, Y).
\]
\end{theorem}

\begin{definition}
\textbf{Коэффициентом корреляции}~$r_{XY}$ называется нормированная ковариация:
\[
r_{XY} = \frac{\Cov(X, Y)}{\sigma_X \sigma_Y}.
\]
Свойства:
\begin{enumerate}
\item $-1 \le r_{XY} \le 1$.
\item $\abs{r_{XY}} = 1 \iff Y = aX + b$ (строгая линейная связь).
\item Если $X$ и $Y$ независимы, то $r_{XY} = 0$.
\end{enumerate}
\end{definition}

\begin{example}[Расчет ковариации на основе матрицы]
Двумерная случайная величина $(X,Y)$ задана матрицей распределения. Найдем ковариацию.
\begin{center}
\begin{tabular}{c|cc|c}
\toprule
$X \setminus Y$ & $Y=0$ & $Y=1$ & $\Prob(X=x_i)$ \\
\midrule
$X=1$ & $0{,}2$ & $0{,}3$ & $0{,}5$ \\
$X=2$ & $0{,}4$ & $0{,}1$ & $0{,}5$ \\
\midrule
$\Prob(Y=y_j)$ & $0{,}6$ & $0{,}4$ & $1{,}0$ \\
\bottomrule
\end{tabular}
\end{center}

\textbf{Решение:}
1. Найдем маргинальные математические ожидания $X$ и $Y$:
\[
\E[X] = 1 \cdot 0{,}5 + 2 \cdot 0{,}5 = 0{,}5 + 1 = 1{,}5.
\]
\[
\E[Y] = 0 \cdot 0{,}6 + 1 \cdot 0{,}4 = 0{,}4.
\]
2. Найдем математическое ожидание произведения $\E[XY]$:
\[
\E[XY] = \sum \sum x_i y_j p_{ij} = 1\cdot 0\cdot 0{,}2 + 1\cdot 1\cdot 0{,}3 + 2\cdot 0\cdot 0{,}4 + 2\cdot 1\cdot 0{,}1 = 0{,}3 + 0{,}2 = 0{,}5.
\]
3. Вычислим ковариацию:
\[
\Cov(X, Y) = \E[XY] - \E[X]\E[Y] = 0{,}5 - 1{,}5 \cdot 0{,}4 = 0{,}5 - 0{,}6 = -0{,}1.
\]
Отрицательная ковариация указывает на наличие обратной линейной зависимости между $X$ и $Y$ (с ростом~$X$ значения~$Y$ имеют тенденцию к снижению).
\end{example}
